    const playHistoryTrail = (timeStamp: number) => {
      if (!map || !trail_points || trail_points.length < 2) return;
      if (!trail_isPlay) {
        trail_preTimeStamp = 0;
        return;
      }
      // 初始时间戳
      if (!trail_preTimeStamp) {
        trail_preTimeStamp = timeStamp;
      }

      const _distance = ((timeStamp - trail_preTimeStamp) / 1000 / 3600) * trail_speed || 0;
      trail_curDistance += _distance;

      // 当前走到的位置
      const line = lineString(trail_points);
      const segment = along(line, trail_curDistance).geometry.coordinates;
      trail_points_even.push(segment);

      // 计算走过的距离的百分比(近似)
      const percent = Math.round((trail_curDistance / trail_distance) * 100);

      // 判断是否已经到终点
      const toDesDistance = length(lineString([segment, trail_points[trail_points.length - 1]]));

      if (toDesDistance > 0.00000001 || percent < 100) {
        // 移动点位
        const point = {
          type: 'Feature',
          properties: {},
          geometry: {
            type: 'Point',
            coordinates: segment,
          },
        };
        const soure_point = map.getSource(TRAIL_MARK_LAYERID);
        if (soure_point) soure_point.setData(point);

        // 判断当前点位是否超出地图范围
        const isPointInMap = map.getBounds().contains(segment);
        if (!isPointInMap) {
          map.jumpTo({ center: segment });
        }
        trail_preTimeStamp = timeStamp;
        requestAnimationFrame(playHistoryTrail);
      } else {
        trail_isPlay = false;
        if (is_run_end_callcallback) {
          trail_end_callback();
          is_run_end_callcallback = false;
        }
      }
    };
