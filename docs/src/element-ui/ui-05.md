# build entry

接上篇，我们创建一个`build-entry.js`，用于**自动生成入口文件`src/index.js` 内容**。

```bash
code build/bin/build-entry.js
```

创建`scripts`字段：

```bash
npm set-script build:entry "node build/bin/build-entry.js"
```



## index.js

在自动生成入口文件（`src/index.js`）之前，我们需要了解文件内容大概情况（**精简版**）：

```js
import Demo from '../packages/demo/index.js';


const components = [
  Demo,
];


const install = function(Vue) {
  
  components.forEach(component => {
    Vue.component(component.name, component);
  });

};

export default {
  install,
  Demo
}
```



## build-entry.js

根据上面精简版，我们推导出字符串模板：

### MAIN_TEMPLATE

```js
var MAIN_TEMPLATE = `
{{include}}

const components = [
{{list}}
];

const install = function(Vue) {

  components.forEach(component => {
    Vue.component(component.name, component);
  });

};

export default {
  install,
{{list}}
};
`;
```

* `{{include}}`：引入组件
* `{{list}}`：组件名称

我们需要动态的生成`{{include}}`和`{{list}}`，并最终输出到`src/index.js`中。要生成这个字符串模板，我们需要借助`json-templater`模块。



### json-templater

[json-templater](https://www.npmjs.com/package/json-templater)模块，是一个通过替换**mustache style** （`{{}}`）的模板生成器。通过命令行，安装依赖：

```bash
yarn add -D json-templater@1.0.4
```

下面是官方demo：

```js
var render = require('json-templater/string');

render('{{xfoo}} {{say.what}}', {
  xfoo: 'yep', 
  say: { what: 'yep' }
});

// output: yep yep
```

是不是非常简单。总结一下：

```js
// 引入依赖的render函数
var render = require('json-templater/string');

/**
 * render接受两个参数：
 * 1. YOUR_TEMPLATE：定义的目标模板，模板中的变量需要使用mustache style，即：`{{}}`
 * 2. YOUR_DATA：模板数据，它是一个对象
render(<YOUR_TEMPLATE>, <YOUR_DATA>)
```



### render

接下来，我们引入依赖，并对目标模板进行渲染：

```js
var render = require('json-templater/string');
var endOfLine = require('os').EOL;

var includeComponentTemplate = [
  'import Dome from \'../packages/dome/index.js\';'
];
var listTemplate = ['Demo'];

var template = render(MAIN_TEMPLATE, {
  include: includeComponentTemplate.join(endOfLine),
  list: listTemplate.join(',' + endOfLine)
});
```

执行`yarn build:entry`后，会输出模板内容：

```bash
import Dome from '../packages/dome/index.js';

const components = [
Demo
];

const install = function(Vue) {

  components.forEach(component => {
    Vue.component(component.name, component);
  });

};

export default {
  install,
Demo
};
```

没错，这就是我们所期望的。



### gen template

按下来，我们需要对`includeComponentTemplate` 和 `listTemplate` 进行动态处理。它们的内容都需要来自于组件信息（**组件名称**和**组件路径**）。

还记得我们在前面，将所有的组件信息通过**json**形式写入`components.json`，统一管理。


## writeFileSync

我们期望的模板已经生成了，接下来就是将模块写入`src/index.js`中，下面是`build/bin/build-entry.js`精简完整版代码：

```js
var Components = require('../../components.json');
var fs = require('fs');
var render = require('json-templater/string');
var uppercamelcase = require('uppercamelcase');
var path = require('path');
var endOfLine = require('os').EOL;

var OUTPUT_PATH = path.join(__dirname, '../../src/index.js');
var IMPORT_TEMPLATE = 'import {{name}} from \'../packages/{{package}}/index.js\';';

var MAIN_TEMPLATE = `/* Automatically generated by './build/bin/build-entry.js' */

{{include}}

const components = [
{{list}}
];

const install = function(Vue) {

  components.forEach(component => {
    Vue.component(component.name, component);
  });

};

export default {
  install,
{{list}}
};
`;

var ComponentNames = Object.keys(Components);

var includeComponentTemplate = [];
var listTemplate = [];

ComponentNames.forEach(name => {
  var componentName = uppercamelcase(name);

  includeComponentTemplate.push(render(IMPORT_TEMPLATE, {
    name: componentName,
    package: name
  }));

  listTemplate.push(`  ${componentName}`);
});

var template = render(MAIN_TEMPLATE, {
  include: includeComponentTemplate.join(endOfLine),
  list: listTemplate.join(',' + endOfLine)
});

fs.writeFileSync(OUTPUT_PATH, template);
console.log('[build entry] DONE:', OUTPUT_PATH);
```

最终，`src/index.js`生成内容如下：

```js
/* Automatically generated by './build/bin/build-entry.js' */

import Loading from '../packages/loading/index.js';
import Demo from '../packages/demo/index.js';

const components = [
  Loading,
  Demo
];

const install = function(Vue) {

  components.forEach(component => {
    Vue.component(component.name, component);
  });

};

export default {
  install,
  Loading,
  Demo
};

```



## 总结

我们通过`json-templater`，借助`components.json`内容生成最终`src/index.js`入口文件的内容。

至此，我们实践了自动创建组件的完整流程（创建组件及组件样式、更新`components.json`、更新入口文件），虽然是一个精简版，但足以给我们开发组件提升不少效率。

关于**精简版**没考虑到的场景有： `this.$loading`、`v-loading`、`icon`等。我们会在后面逐步对其进行补充和完善。
